cmake_minimum_required(VERSION 3.24)
project(Screencast VERSION 0.3.4 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Screencast requires Windows 11 and the MSVC Windows SDK toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(Screencast WIN32 src/main.cpp src/capture.cpp src/audio.cpp src/camera.cpp src/recorder.cpp src/Screencast.rc)
target_compile_definitions(Screencast PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(Screencast PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_include_directories(Screencast PRIVATE assets src)
target_link_options(Screencast PRIVATE /MANIFEST:NO)
set_property(TARGET Screencast PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(Screencast PRIVATE windowsapp d3d11 dxgi dwmapi mfplat mfreadwrite mfuuid mf ole32 oleaut32 uuid avrt propsys comdlg32 comctl32 shlwapi shell32 advapi32 windowscodecs gdiplus)
add_custom_command(TARGET Screencast POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:Screencast>/assets")
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_compile_options(core_tests PRIVATE /W4 /EHsc /utf-8)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(instance_tests WIN32 tests/instance_tests.cpp)
  target_compile_definitions(instance_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(instance_tests PRIVATE /W4 /EHsc /utf-8)
  target_link_libraries(instance_tests PRIVATE shell32 ole32 uuid)
  add_test(NAME single_instance COMMAND instance_tests)
  set_tests_properties(single_instance PROPERTIES TIMEOUT 45)
endif()
install(TARGETS Screencast RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(FILES README.md THIRD_PARTY.md DESTINATION .)
if(SCREENCAST_PAYLOAD)
  configure_file(src/installer.rc.in ${CMAKE_CURRENT_BINARY_DIR}/installer.rc @ONLY)
  add_executable(ScreencastSetup WIN32 src/installer.cpp ${CMAKE_CURRENT_BINARY_DIR}/installer.rc)
  target_compile_definitions(ScreencastSetup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(ScreencastSetup PRIVATE /W4 /EHsc /utf-8 /permissive-)
  target_link_options(ScreencastSetup PRIVATE /MANIFEST:NO)
  set_property(TARGET ScreencastSetup PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(ScreencastSetup PRIVATE windowsapp ole32 oleaut32 uuid shell32 propsys advapi32 dwmapi gdiplus windowscodecs)
endif()
