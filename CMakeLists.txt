cmake_minimum_required(VERSION 3.24)
project(Screencast VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Screencast requires Windows 11 and the MSVC Windows SDK toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(Screencast WIN32 src/main.cpp src/capture.cpp src/audio.cpp src/camera.cpp src/recorder.cpp src/Screencast.rc)
target_compile_definitions(Screencast PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(Screencast PRIVATE /W4 /EHsc /utf-8 /permissive-)
set_property(TARGET Screencast PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(Screencast PRIVATE windowsapp d3d11 dxgi dwmapi mfplat mfreadwrite mfuuid mf ole32 oleaut32 uuid avrt propsys comdlg32 comctl32 shlwapi shell32 advapi32 windowscodecs)
add_custom_command(TARGET Screencast POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:Screencast>/assets")
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_compile_options(core_tests PRIVATE /W4 /EHsc /utf-8)
  add_test(NAME core_tests COMMAND core_tests)
endif()
install(TARGETS Screencast RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(FILES README.md THIRD_PARTY.md DESTINATION .)
